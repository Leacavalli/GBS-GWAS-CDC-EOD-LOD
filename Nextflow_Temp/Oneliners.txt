path_nextflow_dir=$(echo "/n/holylfs05/LABS/hanage_lab/Lab/hsphfs1/lcavalli/GBS-GWAS-CDC-EOD-LOD/NEXTFLOW_PIPELINE")

mkdir $path_nextflow_dir/8.ANNOTATION/Filter_out_QC


for i in $path_nextflow_dir/9.QUAST/*.quast.out;
do
  # Extract the necessary values from the QUAST output file
  assembly=$(grep "Assembly" $i/report.txt | awk '{print $2}')
  n50=$(grep -w "N50" $i/report.txt | awk '{print $2}')
  gc=$(grep -w "GC (%)" $i/report.txt | awk '{print $3}')
  total_length=$(grep -w "Total length" $i/report.txt | grep -v "(>= " | awk '{print $3}')
  contigs=$(grep "^# contigs " $i/report.txt | tail -1 | awk '{print $3}')
  # Extract the necessary values from the FastANI output file
  ANI=$(cat $path_nextflow_dir/8.FastANI/$assembly\.fastani.out| awk '{print $3}')

  # Check the conditions
  if (( $(echo "$ANI < 95" | bc -l) )) ||[ "$n50" -lt  30000 ] || [ "$contigs" -gt 250 ]  || (( $(echo "$gc <= 30" | bc -l) )) || (( $(echo "$gc >= 40" | bc -l) )) || [ "$total_length" -lt 1700000 ] || [ "$total_length" -gt 2400000 ] ; then
    mv $path_nextflow_dir/6.SNIPPY/$assembly $path_nextflow_dir/6.SNIPPY/Filter_out_QC
    mv $path_nextflow_dir/7.ASSEMBLIES/$assembly\.fasta $path_nextflow_dir/7.ASSEMBLIES/Filter_out_QC
    mv $path_nextflow_dir/8.ANNOTATION/$assembly\.gff $path_nextflow_dir/8.ANNOTATION/Filter_out_QC
  fi
done

#Panda bash script 
import pandas as pd

# Load your SC classifications
sc_data = pd.read_csv("sc_classifications.csv", index_col=0)
samples = sc_data.index

# Create an empty similarity matrix
similarity_matrix = pd.DataFrame(0, index=samples, columns=samples)

# Fill the matrix: 1 if in the same SC, 0 otherwise
for sc in sc_data['SC'].unique():
    members = sc_data[sc_data['SC'] == sc].index
    for member1 in members:
        for member2 in members:

#Panda bash script 
import pandas as pd

# Load your SC classifications
sc_data = pd.read_csv("sc_classifications.csv", index_col=0)
samples = sc_data.index

# Create an empty similarity matrix
similarity_matrix = pd.DataFrame(0, index=samples, columns=samples)

# Fill the matrix: 1 if in the same SC, 0 otherwise
for sc in sc_data['SC'].unique():
    members = sc_data[sc_data['SC'] == sc].index
    for member1 in members:
        for member2 in members:
            similarity_matrix.loc[member1, member2] = 1

# Save the matrix
similarity_matrix.to_csv("sc_similarity_matrix.tsv", sep="\t")
